let db = null;
let cart = {};
let ready = false;

const initialProducts = [

  [7,'Logitech Mouse','Աքսեսուարներ',18000,20,'https://images.unsplash.com/photo-1615663245857-ac93bb7c39e7?w=400&q=80'],

async function init() {
  try {
    const SQL = await initSqlJs({
      locateFile: file =>
        'https://cdnjs.cloudflare.com/ajax/libs/sql.js/1.10.3/' + file
    });

    db = new SQL.Database();

    db.run(`
      CREATE TABLE products(
        id INTEGER PRIMARY KEY,
        name TEXT NOT NULL,
        category TEXT,
        price REAL NOT NULL,
        stock INTEGER DEFAULT 0,
        emoji TEXT
      );

      CREATE TABLE customers(
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        name TEXT NOT NULL
      );

      CREATE TABLE orders(
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        customer_id INTEGER,
        total REAL NOT NULL,
        created_at TEXT DEFAULT CURRENT_TIMESTAMP,
        FOREIGN KEY(customer_id) REFERENCES customers(id)
      );

      CREATE TABLE order_items(
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        order_id INTEGER,
        product_id INTEGER,
        quantity INTEGER,
        unit_price REAL,
        FOREIGN KEY(order_id) REFERENCES orders(id),
        FOREIGN KEY(product_id) REFERENCES products(id)
      );
    `);

    const stmt = db.prepare(
      'INSERT INTO products VALUES (?, ?, ?, ?, ?, ?)'
    );
    initialProducts.forEach(p => stmt.run(p));
    stmt.free();

    ready = true;
    renderProducts();
    sample('SELECT * FROM products;');
  } catch (e) {
    document.getElementById('products').textContent =
      'Չհաջողվեց բեռնել SQL գրադարանը։ Ստուգիր ինտերնետային կապը։';
    document.getElementById('result').textContent = e.message;
  }
}

function money(n) {
  return Number(n).toLocaleString('en-US') + ' ֏';
}

function renderProducts() {
  if (!ready) return;
  const term = document.getElementById('search').value.toLowerCase();
  const cat = document.getElementById('category').value;

  const stmt = db.prepare(
    'SELECT * FROM products WHERE lower(name) LIKE ? AND category LIKE ? ORDER BY id'
  );
  stmt.bind(['%' + term + '%', cat || '%']);

  let html = '';
  while (stmt.step()) {
    const p = stmt.getAsObject();
    html += `<article class="card">
      <div class="pic"><img src="${escapeHTML(p.emoji || '')}" alt="${escapeHTML(p.name)}"></div>
      <small>${escapeHTML(p.category || '')} · Պահեստում՝ ${p.stock}</small>
      <h3>${escapeHTML(p.name)}</h3>
      <div class="price">${money(p.price)}</div>
      <button onclick="addToCart(${p.id})" ${p.stock <= 0 ? 'disabled' : ''}>Ավելացնել զամբյուղ</button>
    </article>`;
  }
  stmt.free();

  document.getElementById('products').innerHTML =
    html || '<p>Ապրանքներ չեն գտնվել։</p>';
}

function escapeHTML(s) {
  return String(s).replace(/[&<>"']/g, c => ({
    '&':'&amp;', '<':'&lt;', '>':'&gt;',
    '"':'&quot;', "'":'&#39;'
  })[c]);
}

function addToCart(id) {
  const row = db.exec('SELECT stock FROM products WHERE id=' + Number(id));
  if (!row.length || !row[0].values.length) return;
  const stock = row[0].values[0][0];
  if ((cart[id] || 0) >= stock) {
    alert('Պահեստում այսքան ապրանք չկա։');
    return;
  }
  cart[id] = (cart[id] || 0) + 1;
  renderCart();
}

function renderCart() {
  let total = 0, count = 0, html = '';
  for (const id of Object.keys(cart)) {
    const rows = db.exec('SELECT name,price FROM products WHERE id=' + Number(id));
    if (!rows.length || !rows[0].values.length) continue;
    const [name, price] = rows[0].values[0];
    const qty = cart[id];
    total += price * qty;
    count += qty;
    html += `<p>${escapeHTML(name)} × ${qty} — ${money(price * qty)}
      <button class="secondary" onclick="removeItem(${id})">−</button></p>`;
  }
  document.getElementById('cartItems').innerHTML = html || 'Զամբյուղը դատարկ է։';
  document.getElementById('total').textContent = money(total);
  document.getElementById('count').textContent = count;
}

function removeItem(id) {
  cart[id]--;
  if (cart[id] <= 0) delete cart[id];
  renderCart();
}

function checkout() {
  if (!ready) return;
  const name = document.getElementById('customer').value.trim();
  const message = document.getElementById('orderMessage');

  if (!name) {
    message.textContent = 'Մուտքագրիր հաճախորդի անունը։';
    return;
  }
  if (!Object.keys(cart).length) {
    message.textContent = 'Զամբյուղը դատարկ է։';
    return;
  }

  try {
    db.run('BEGIN TRANSACTION');
    db.run('INSERT INTO customers(name) VALUES (?)', [name]);
    const customerId = db.exec('SELECT last_insert_rowid()')[0].values[0][0];

    let total = 0;
    for (const id of Object.keys(cart)) {
      const product = db.exec(
        'SELECT price, stock FROM products WHERE id=' + Number(id)
      )[0].values[0];
      if (cart[id] > product[1]) throw new Error('Ապրանքի քանակը բավարար չէ։');
      total += product[0] * cart[id];
    }

    db.run('INSERT INTO orders(customer_id,total) VALUES (?,?)',
      [customerId,total]);
    const orderId = db.exec('SELECT last_insert_rowid()')[0].values[0][0];

    for (const id of Object.keys(cart)) {
      const product = db.exec(
        'SELECT price FROM products WHERE id=' + Number(id)
      )[0].values[0];
      db.run(
        'INSERT INTO order_items(order_id,product_id,quantity,unit_price) VALUES (?,?,?,?)',
        [orderId,Number(id),cart[id],product[0]]
      );
      db.run('UPDATE products SET stock=stock-? WHERE id=?',
        [cart[id],Number(id)]);
    }

    db.run('COMMIT');
    cart = {};
    renderCart();
    renderProducts();
    message.textContent = 'Պատվերը հաջողությամբ գրանցվեց։ Պատվերի համար՝ ' + orderId;
    sample('SELECT o.id, c.name AS customer, o.total, o.created_at FROM orders o JOIN customers c ON c.id=o.customer_id;');
  } catch(e) {
    try { db.run('ROLLBACK'); } catch (_) {}
    message.textContent = 'Սխալ՝ ' + e.message;
  }
}

function sample(sql) {
  document.getElementById('query').value = sql;
  runSQL();
}

function showSchema() {
  sample("SELECT name, sql FROM sqlite_master WHERE type='table' ORDER BY name;");
}

function runSQL() {
  if (!ready) {
    document.getElementById('result').textContent = 'Տվյալների բազան բեռնվում է...';
    return;
  }
  const out = document.getElementById('result');
  out.innerHTML = '';

  try {
    const results = db.exec(document.getElementById('query').value);
    if (!results.length) {
      out.textContent = 'Հրամանը կատարվեց։ Արդյունք վերադարձնող տվյալներ չկան։';
    }

    results.forEach(r => {
      const table = document.createElement('table');
      const thead = document.createElement('thead');
      const tr = document.createElement('tr');

      r.columns.forEach(col => {
        const th = document.createElement('th');
        th.textContent = col;
        tr.appendChild(th);
      });
      thead.appendChild(tr);
      table.appendChild(thead);

      const tbody = document.createElement('tbody');
      r.values.forEach(row => {
        const tr = document.createElement('tr');
        row.forEach(value => {
          const td = document.createElement('td');
          td.textContent = value === null ? 'NULL' : value;
          tr.appendChild(td);
        });
        tbody.appendChild(tr);
      });
      table.appendChild(tbody);
      out.appendChild(table);
    });

    renderProducts();
  } catch(e) {
    out.textContent = 'SQL Error: ' + e.message;
  }
}
